  number line/.style={

  },
  number tick/.style={

  },

    -{Stealth[length=2.4mm]},
    line width=0.8pt
  },
  grid line/.style={
    line width=0.3pt,
    draw=gray!45
  },
  geometry line/.style={
    line width=0.8pt
  },
  point mark/.style={
    circle,
    fill=black,
    inner sep=1.2pt
  }
}
