\newcommand{\NumberLine}[4]{%
  \begin{tikzpicture}[scale=0.78]
    \draw[course axis] (#1,0) -- (#2,0);
    \foreach \x in {#3} {
      \draw (\x,0.08) -- (\x,-0.08);
      \node[below] at (\x,-0.08) {\small \x};
    }
    #4
  \end{tikzpicture}
}

\newcommand{\BlankNumberLine}{%
  \begin{tikzpicture}[scale=0.78]
    \draw[course axis] (-5.2,0) -- (5.6,0);

      \draw (\x,0.08) -- (\x,-0.08);
    }
    \node[below] at (0,-0.08) {\small 0};
  \end{tikzpicture}
}
